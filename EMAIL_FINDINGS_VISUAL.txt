3. EMAIL TRIGGER FLOWCHART
═══════════════════════════════════════════════════════════════════════════════

                         USER SIGNUP → POST /api/auth/signup
                                │
                                ├──► CREATE Profile
                                ├──► CREATE LoginLog
                                └──► SEND Welcome Email
                                      ├─ Template: welcome.tsx
                                      ├─ Recipient: user.email
                                      └─ Content: Get Started Guide


                         TRY FREE TRIAL → POST /api/trial
                                │
                                ├──► UPDATE Subscription (status: trialing)
                                └──► SEND Trial Activation Email
                                      ├─ Template: trial-start.tsx
                                      ├─ Recipient: user.email
                                      └─ Content: Features + Countdown


                         [DAILY CRON JOB] → Check Trial Expirations
                                │
                                ├──► IF expires in 24 hours
                                │    └──► SEND Trial Warning Email
                                │         ├─ Template: trial-warning.tsx
                                │         └─ Content: "Upgrade Now"
                                │
                                └──► IF expired
                                     ├──► UPDATE Subscription (plan: free)
                                     └──► SEND Trial Expired Email
                                          ├─ Template: trial-expired.tsx
                                          └─ Content: "Upgrade to Pro"


                         PAYMENT RECEIVED → POST /api/stripe/webhook
                                │
                                ├──► Case: checkout.session.completed
                                │    ├──► UPDATE Subscription (plan: pro/design_partner)
                                │    └──► SEND Subscription Activated Email
                                │
                                └──► Case: customer.subscription.updated
                                     ├──► UPDATE Subscription
                                     └──► SEND Subscription Updated Email
                                          (plan changed, new features)


                         CANCEL SUBSCRIPTION → POST /api/stripe/webhook
                                │
                                ├──► Case: customer.subscription.deleted
                                │    ├──► UPDATE Subscription (status: canceled)
                                │    └──► SEND Subscription Canceled Email
                                │         ├─ Content: "Sorry to see you go"
                                │         └─ Content: "Come back offer"


                         SUPPORT REQUEST → POST /api/support
                                │
                                ├──► CREATE SupportTicket
                                ├──► SEND Ticket Confirmation Email
                                │    ├─ Template: support-confirmation.tsx
                                │    ├─ Recipient: requester.email
                                │    └─ Content: "Ticket #123 received"
                                │
                                └──► SEND New Ticket Alert
                                     ├─ Template: admin-alert.tsx
                                     ├─ Recipient: ADMIN_EMAILS
                                     └─ Content: "New ticket from user@domain.com"


                         ADMIN UPDATES TICKET → PATCH /api/admin/support
                                │
                                ├──► UPDATE SupportTicket (status: in_progress/resolved)
                                └──► SEND Status Update Email
                                     ├─ Template: support-update.tsx
                                     ├─ Recipient: original_requester.email
                                     └─ Content: "Your ticket is now resolved"
